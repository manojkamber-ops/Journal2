    24→    24→// ── Seeded PRNG (mulberry32) ─────────────────────────────────────────────────
    25→    25→function mulberry32(seed: number) {
    26→    26→  return function () {
    27→    27→    let t = (seed += 0x6d2b79f5);
    28→    28→    t = Math.imul(t ^ (t >>> 15), t | 1);
    29→    29→    t ^= t + Math.imul(t ^ (t >>> 7), t | 61);
    30→    30→    return ((t ^ (t >>> 14)) >>> 0) / 4294967296;
    31→    31→  };
    32→    32→}

    38→    38→function weightedPick<T>(items: T[], weights: number[]): T {
    39→    39→  const total = weights.reduce((a, b) => a + b, 0);
    40→    40→  let v = R() * total;
    41→    41→  for (let i = 0; i < items.length; i++) {
    42→    42→    v -= weights[i];
    43→    43→    if (v <= 0) return items[i];
    44→    44→  }
    45→    45→  return items[items.length - 1];
    46→    46→}

   131→   131→function titleCase(s: string): string {
   132→   132→  const words = s.split(/\s+/);
   133→   133→  return words
   134→   134→    .map((w, i) => {
   135→   135→      const isEdge = i === 0 || i === words.length - 1;
   136→   136→      return w
   137→   137→        .split('-')
   138→   138→        .map(p => {
   139→   139→          if (p.length >= 2 && p === p.toUpperCase()) return p; // ALL-CAPS: CNN, PCB, BLE, R-CNN
   140→   140→          if (!/^[a-z0-9]+$/.test(p)) return p; // mixed case brands unchanged: FastText, YOLOv8, ParsBERT
   141→   141→          const lower = p.toLowerCase();
   142→   142→          if (!isEdge && SMALL.has(lower)) return lower;
   143→   143→          return lower.charAt(0).toUpperCase() + lower.slice(1);
   144→   144→        })
   145→   145→        .join('-');
   146→   146→    })
   147→   147→    .join(' ');
   148→   148→}

   150→   150→function aAn(word: string, upper = true): string {
   151→   151→  const c = word.charAt(0);
   152→   152→  let vowel = ['a', 'e', 'i', 'o', 'u'].includes(c.toLowerCase());
   153→   153→  if (c === 'U') vowel = false; // uppercase U- prefix (U-Net) is pronounced "you"
   154→   154→  const art = vowel ? 'an' : 'a';
   155→   155→  return upper ? art.charAt(0).toUpperCase() + art.slice(1) : art;
   156→   156→}

   600→   600→function shufflePick(pool: string[], n: number): string[] {
   601→   601→  const arr = [...pool];
   602→   602→  const out: string[] = [];
   603→   603→  while (out.length < n && arr.length) out.push(arr.splice(Math.floor(R() * arr.length), 1)[0]);
